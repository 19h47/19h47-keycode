@19h47/keycode

Simple keycode helper

Named constants for legacy KeyboardEvent.keyCode values.

Usage

import { ENTER, ESCAPE, ARROW_DOWN } from '@19h47/keycode';

document.addEventListener('keydown', (event) => {
	if (event.keyCode === ENTER) {
		// …
	}

	if (event.keyCode === ESCAPE) {
		// …
	}
});
